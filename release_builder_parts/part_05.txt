def sig_results(results, kind):

def result_list(sig):

def collapse_shear(rows):
    # input unique exact rows. Collapse successive numeric heights with same selector/result to ranges.
    groups={}
    for r in rows:
        h=str(r['height_mm'])
        key=(r['moment_class'],r['shear_class'],r['concrete_min'],r['cover'],
             json.dumps(r['results'],sort_keys=True,ensure_ascii=False,separators=(',',':')),
             r.get('compression_transfer',''), tuple(r.get('notes') or []), tuple(r.get('source_pages') or []))
        groups.setdefault(key,[]).append(h)
    out=[]
    for key, hs in groups.items():
        numeric=sorted({int(h) for h in hs if h.isdigit()})
        nonnum=sorted({h for h in hs if not h.isdigit()})
        ranges=[]
        if numeric:
            start=prev=numeric[0]
            # heights are typically 5 mm; respect any consecutive 5mm series; isolated boundary points stay isolated.
            for h in numeric[1:]:
                if h-prev==5:
                    prev=h
                else:
                    ranges.append((start,prev)); start=prev=h
            ranges.append((start,prev))
        for a,b in ranges:
            rr={
                'moment_class':key[0],'shear_class':key[1],'concrete_min':key[2],'cover':key[3],
                'height_min':str(a),'height_max':str(b),
                'results':json.loads(key[4]),'compression_transfer':key[5],
                'source_pages':list(key[7]),
            }
            if key[6]: rr['notes']=list(key[6])
            out.append(rr)
        for h in nonnum:
            rr={
                'moment_class':key[0],'shear_class':key[1],'concrete_min':key[2],'cover':key[3],
                'height_min':h,'height_max':h,'results':json.loads(key[4]),'compression_transfer':key[5],
                'source_pages':list(key[7]),
            }
            if key[6]: rr['notes']=list(key[6])
            out.append(rr)
    return out

def compact_package(pkg, matrix_types):
    new={'schema_version':3,'catalog':pkg['catalog'],'families':[],'shared_matrices':{}}
    matrix_signatures={}
    for fam in pkg['families']:
        f={k:v for k,v in fam.items() if k!='records'}
        recs=fam.get('records',[])
        if fam['type'] not in matrix_types:
            f['records']=recs
            f['backend']='records'
            new['families'].append(f); continue
        f['backend']='matrix'
        moments={}
        shear_exact={}
        # Every source record is a valid join of a moment row and a shear row.
        for r in recs:
            mres=[x for x in r.get('results',[]) if x.get('kind')=='moment']
            sres=[x for x in r.get('results',[]) if x.get('kind')=='shear']
            other=[x for x in r.get('results',[]) if x.get('kind') not in {'moment','shear'}]
            if mres or other:
                key=(r['moment_class'],r['concrete_min'],r['cover'],str(r['height_mm']),
                     json.dumps(mres+other,sort_keys=True,ensure_ascii=False,separators=(',',':')),
                     r.get('compression_transfer',''))
                if key not in moments:
                    moments[key]={
                        'moment_class':r['moment_class'],'concrete_min':r['concrete_min'],'cover':r['cover'],
                        'height_mm':str(r['height_mm']),'results':mres+other,
                        'compression_transfer':r.get('compression_transfer',fam.get('compression_transfer','')),
                        'source_pages':r.get('source_pages',fam.get('source_pages',[])),
                    }
            if sres:
                key=(r['moment_class'],r['shear_class'],r['concrete_min'],r['cover'],str(r['height_mm']),
                     json.dumps(sres,sort_keys=True,ensure_ascii=False,separators=(',',':')),
                     r.get('compression_transfer',''), tuple(r.get('notes') or []))
                if key not in shear_exact:
                    shear_exact[key]={
                        'moment_class':r['moment_class'],'shear_class':r['shear_class'],'concrete_min':r['concrete_min'],
                        'cover':r['cover'],'height_mm':str(r['height_mm']),'results':sres,
                        'compression_transfer':r.get('compression_transfer',fam.get('compression_transfer','')),
                        **({'notes':r.get('notes')} if r.get('notes') else {})
                    }
        moment_rows=list(moments.values())
        shear_rows=collapse_shear(list(shear_exact.values()))
        # HVS/WOS and BH/WU/BHS/WUS use the same numerical design tables.
        # Store identical matrices only once and let several family metadata records reference them.
        signature=hashlib.sha256(json.dumps([moment_rows,shear_rows],ensure_ascii=False,sort_keys=True,separators=(',',':')).encode()).hexdigest()
        if signature not in matrix_signatures:
            ref=f'matrix_{len(matrix_signatures)+1:02d}'
            matrix_signatures[signature]=ref
            new['shared_matrices'][ref]={'moment_rows':moment_rows,'shear_rows':shear_rows}
        else:
            ref=matrix_signatures[signature]
        f['matrix_ref']=ref
        f['designation_style']='maxfrank_matrix'
        new['families'].append(f)
    return new

def save(pkg,out):
    raw=json.dumps(pkg,ensure_ascii=False,separators=(',',':')).encode()
    text=base64.b64encode(gzip.compress(raw,9)).decode()
    Path(out).write_text(text)
    return len(raw),len(text),hashlib.sha256(text.encode()).hexdigest()
