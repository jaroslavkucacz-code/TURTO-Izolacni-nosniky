def parse_eccentric(pdf,pno,typ,insulation,series,concrete):
    tables=pdf.pages[pno-1].extract_tables()
    fam=family_common(series,typ,insulation,'Smykové prvky – excentrické připojení','přídavný moment od excentrického připojení',[pno],'dle základního smykového prvku')
    covers=['C30','C35','C50']
    for t in tables[:2]:
        names=[]
        for x in t[0][3:]:
            if not clean(x):continue
            # keep first designation, rows apply to ± too
            s=clean(x).split(' ')[0].split('\n')[0]
            # pdfplumber clean turned newline to space; take before second same prefix by regex
            m=re.search(r'((?:VM|VXL)\s*Z?\s*\d+(?:-K)?|(?:VM|VXL)\d+(?:-K)?)',clean(x).replace(' ',''))
            if m:s=m.group(1).replace(' ','')
            names.append(s)
        for row in t[5:]:
            ranges=[clean(row[0]),clean(row[1]),clean(row[2])]
            for ci,(cover,rg) in enumerate(zip(covers,ranges)):
                # store range string rather than every 5 mm: this is an auxiliary table
                for j,name in enumerate(names):
                    cell=clean(row[3+j]) if 3+j<len(row) else ''
                    res=result_obj('delta_m_ed','ΔMEd','other',cell,'kNm/element')
                    if not res:continue
                    src=f'{name}-excentricita-{cover}-h{rg}'
                    fam['records'].append({
                        'moment_class':name,'shear_class':'ΔMEd','concrete_min':concrete,'cover':cover,'height_mm':rg,
                        'designation':f'Egcobox® {src} [{concrete}]','aliases':[src,name],
                        'results':[res],'source_pages':[pno],'insulation_thickness_mm':insulation,'compression_transfer':'dle základního smykového prvku'
                    })
    return fam

def parse_interaction_family(pdf,pno,typ,insulation,series,which):
    tables=pdf.pages[pno-1].extract_tables(); geom=tables[0]
    names=[clean(x) for x in geom[0][1:] if clean(x)]
    # page has C20 table1 and C25 table2
    fam=family_common(series,typ,insulation,'Parapety / atiky','interakční tabulkové hodnoty N-Rd / M-Rd a smyk',[pno],'tlačené ocelové pruty')
    for concrete,ti in [('C20/25',1),('C25/30',2)]:
        t=tables[ti]
        # rows 2 until row before V header; each product has N,M pair
        vrow=next(i for i,r in enumerate(t) if any('V [kN/element]' in clean(x) for x in r))
        shear=t[vrow+1]
        for j,name in enumerate(names):
            # geometry as variant in cover/height
            h=clean(geom[2][1+j]) or clean(geom[2][1])
            w=clean(geom[3][1+j]) or clean(geom[3][1])
            vr=result_obj('v_rd_z','VRd','shear',shear[1+2*j] if 1+2*j<len(shear) else '','kN/element')
            for ri,row in enumerate(t[2:vrow]):
                ncell=clean(row[1+2*j]) if 1+2*j<len(row) else ''
                mcell=clean(row[2+2*j]) if 2+2*j<len(row) else ''
                nr=result_obj('n_rd','NRd','normal',ncell,'kN/element')
                mr=result_obj('m_rd','MRd','moment',mcell,'kNm/element')
                if not (nr or mr):continue
                variant=f'w {w}' if w else '—'
                src=f'{name}-N{ncell}-h{h}'.replace(' ','')
                fam['records'].append({
                    'moment_class':name,'shear_class':f'Interakce {ri+1}','concrete_min':concrete,'cover':variant,'height_mm':h,
                    'designation':f'Egcobox® {name} | N={ncell}, M={mcell} [{concrete}]','aliases':[name,src],
                    'results':[x for x in [nr,mr,vr] if x], 'source_pages':[pno],
                    'insulation_thickness_mm':insulation,'compression_transfer':'tlačené ocelové pruty'
                })
    return fam
