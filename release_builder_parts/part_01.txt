def clean(x):
    if x is None: return ''
    return re.sub(r'\s+',' ',str(x)).strip()

def fnum(x):
    s=clean(x).replace('−','-').replace('–','-').replace(' ','').replace(',','.')
    s=s.replace('±','')
    if s in {'','-','—'}: return None
    try:return float(s)
    except:return None

def result_obj(key,label,kind,cell,unit,note=None):
    s=clean(cell).replace('−','-').replace('–','-').replace(' ','')
    if not s or s in {'-','—'}: return None
    d={'key':key,'label':label,'kind':kind,'unit':unit}
    if note: d['note']=note
    if s.startswith('±'):
        v=fnum(s)
        if v is None:return None
        d['positive']=abs(v); d['negative']=-abs(v); return d
    # +x/-y or x/-y
    m=re.match(r'^\+?([0-9.,]+)\s*/\s*-([0-9.,]+)$', s)
    if m:
        d['positive']=float(m.group(1).replace(',','.')); d['negative']=-float(m.group(2).replace(',','.')); return d
    v=fnum(s)
    if v is None:return None
    d['value']=v; return d

def parse_range(s):
    s=clean(s).replace('–','-').replace('—','-').replace(' ','')
    m=re.fullmatch(r'(\d+)-(\d+)',s)
    if m:return int(m.group(1)),int(m.group(2))
    m=re.fullmatch(r'≥(\d+)',s)
    if m:return int(m.group(1)),9999
    if s.isdigit():
        v=int(s); return v,v
    return None

def in_range(h,s):
    r=parse_range(s)
    return bool(r and r[0] <= int(h) <= r[1])

def iter_range(s, step=5):
    r=parse_range(s)
    if not r:return []
    a,b=r
    if b>=9999: return [str(a)]
    # catalog heights use 5 mm increments
    start=((a+step-1)//step)*step
    vals=list(range(start,b+1,step))
    if a not in vals and a%step: vals.insert(0,a)
    return [str(x) for x in vals]

def product_names_main(table,prefix):
    joined=' '.join(clean(x) for x in table[0][4:] if clean(x))
    # MM / MXL with optional K and ±
    pat=rf'\b{re.escape(prefix)}\d+(?:-K)?(?:±)?\b'
    names=re.findall(pat,joined)
    # remove duplicates preserving order
    out=[]
    for n in names:
        if n not in out:out.append(n)
    # Some MAX FRANK tables contain an additional trailing geometry/header column
    # (for example slab/balcony notes). The product designations are therefore
    # the reliable source for the number of capacity columns; downstream parsers
    # deliberately read only the first len(out) product-value cells.
    if not out:
        raise RuntimeError(f'no product names found: header={joined}')
    return out
